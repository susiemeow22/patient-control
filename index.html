<script>

    let patients = JSON.parse(localStorage.getItem('patients') || '[]');

    let isPrivate = false;


    function togglePrivacy() {

        isPrivate = !isPrivate;

        document.getElementById('privacyBtn').innerText = isPrivate ? '🙈 隐藏' : '👁️ 隐私';

        render();

    }


    function render() {

        const kw = document.getElementById('searchInput').value.toLowerCase();

        const filtered = patients.filter(p => p.name.includes(kw) || p.bedNo.includes(kw) || p.diagnosis.includes(kw));

        filtered.sort((a, b) => a.bedNo.localeCompare(b.bedNo));


        const listDiv = document.getElementById('listView');

        listDiv.innerHTML = filtered.map((p, i) => `

            <div class="card" onclick="openModal(${patients.indexOf(p)})">

                <div class="card-row">

                    <span class="${isPrivate ? 'private-blur' : ''}"><b>${p.name}</b></span>

                    <span class="bed-tag">${p.bedNo} 床</span>

                </div>

                <div style="font-size:13px; color:#666;">${p.diagnosis}</div>

            </div>

        `).join('');


        const tbody = document.getElementById('tableBody');

        tbody.innerHTML = filtered.map(p => `

            <tr>

                <td>${p.bedNo}</td>

                <td class="${isPrivate ? 'private-blur' : ''}">${p.name}</td>

                <td>${p.dept}</td>

                <td>${p.diagnosis}</td>

                <td>${p.dateIn}</td>

            </tr>

        `).join('');

    }


    function switchView(view) {

        document.getElementById('listView').style.display = view === 'list' ? 'block' : 'none';

        document.getElementById('tableView').style.display = view === 'table' ? 'block' : 'none';

        document.getElementById('tabList').className = 'tab' + (view === 'list' ? ' active' : '');

        document.getElementById('tabTable').className = 'tab' + (view === 'table' ? ' active' : '');

    }


    function openModal(index = -1) {

        currentEditIndex = index;

        document.getElementById('modal').style.display = 'block';

        const fields = ['bedNo','dept','name','ageId','diagnosis','dateIn','datePre','meds','plan','todos'];

        if (index > -1) {

            fields.forEach(f => document.getElementById(f).value = patients[index][f] || '');

            document.getElementById('delBtn').style.display = 'block';

        } else {

            fields.forEach(f => document.getElementById(f).value = '');

            document.getElementById('delBtn').style.display = 'none';

        }

    }


    function closeModal() { document.getElementById('modal').style.display = 'none'; }


    function savePatient() {

        const p = {};

        ['bedNo','dept','name','ageId','diagnosis','dateIn','datePre','meds','plan','todos'].forEach(f => {

            p[f] = document.getElementById(f).value;

        });

        if (!p.name || !p.bedNo) return alert('必填未填');

        if (currentEditIndex > -1) patients[currentEditIndex] = p;

        else patients.push(p);

        localStorage.setItem('patients', JSON.stringify(patients));

        render();

        closeModal();

    }


    function deletePatient() {

        if(confirm('确定删除？')) {

            patients.splice(currentEditIndex, 1);

            localStorage.setItem('patients', JSON.stringify(patients));

            render(); closeModal();

        }

    }


    function openSettings() { document.getElementById('settingsModal').style.display = 'block'; }


    function exportData() {

        const data = localStorage.getItem('patients');

        navigator.clipboard.writeText(data);

        alert('加密数据已复制到剪贴板，请去备忘录粘贴保存。');

    }


    function importData() {

        const data = prompt('请粘贴以前导出的数据字符串：');

        if (data) {

            localStorage.setItem('patients', data);

            patients = JSON.parse(data);

            render();

            alert('导入成功！');

        }

    }


    function clearAll() {

        if(confirm('警告：这将永久删除手机上存储的所有病人资料！')) {

            localStorage.clear();

            patients = [];

            render();

        }

    }


    render();

</script>
